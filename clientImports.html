// Calculate and display quote based on current form state function calcDHLCost() { try { const countryCode = (document.getElementById('receiverCountry') || {}).value || ''; const weight = parseFloat(document.getElementById('actualWeight')?.value || 0) || 0; const volW = parseFloat(document.getElementById('chargeWeight')?.value || 0) || 0; const chargeWt = Math.max(weight, volW) || weight; if (!countryCode || !chargeWt) return; // Find zone for selected country const zoneEntry = ZONES.find(z => z.code === countryCode); if (!zoneEntry) return; const zone = zoneEntry.zone; const zoneIdx = zone - 1; // zones 1-9 map to index 0-8 // Determine direction and type const direction = (document.getElementById('directionExport')?.checked) ? 'export' : 'import'; const isDoc = (document.getElementById('typeDoc')?.checked); const table = (direction === 'import' ? RATES_IMPORT : RATES_EXPORT)[isDoc ? 'documents' : 'nondocuments']; if (!table) return; // Find closest weight bracket const sorted = [...table].sort((a,b) => a.kg - b.kg); const row = sorted.find(r => r.kg >= chargeWt) || sorted[sorted.length-1]; if (!row) return; const baseCost = row.rates[zoneIdx]; const finalPrice = Math.ceil(baseCost * 1.45); // 45% margin // Display in cost section let costEl = document.getElementById('dhlCostDisplay'); if (!costEl) { costEl = document.createElement('div'); costEl.id = 'dhlCostDisplay'; costEl.style.cssText = 'margin:12px 0;padding:14px 16px;background:linear-gradient(135deg,#0d2137,#0a2f57);border-radius:10px;color:#fff;display:flex;justify-content:space-between;align-items:center'; const rateCard = document.getElementById('costCard') || document.querySelector('.rate-card'); if (rateCard) rateCard.insertAdjacentElement('afterend', costEl); } costEl.innerHTML = `
Estimated Price (Excl. VAT)
R ${finalPrice.toLocaleString()}
Zone ${zone} · ${chargeWt}kg · ${direction} · ${isDoc?'Document':'Non-Document'}
Cost: R ${Math.ceil(baseCost).toLocaleString()}
Margin 45%
`; } catch(e) { console.error('DHL calc error:', e); } } // Auto-calculate when weight or country changes document.addEventListener('DOMContentLoaded', () => { ['actualWeight','chargeWeight','receiverCountry'].forEach(id => { const el = document.getElementById(id); if (el) el.addEventListener('change', calcDHLCost); if (el) el.addEventListener('input', calcDHLCost); }); ['directionExport','directionImport','typeDoc','typeNonDoc'].forEach(id => { const el = document.getElementById(id); if (el) el.addEventListener('change', calcDHLCost); }); });
